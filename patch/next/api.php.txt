/* ---------------- storage ---------------- */
function data_dir(): string {
    static $dir = null;
    if ($dir) return $dir;
    $cands = [dirname(__DIR__, 2) . '/.uroosh-data-' . basename(__DIR__), __DIR__ . '/_data'];
    foreach ($cands as $d) {
        if ((is_dir($d) || @mkdir($d, 0700, true)) && is_writable($d)) {
            if (strpos($d, __DIR__) === 0) {
                if (!file_exists("$d/.htaccess")) @file_put_contents("$d/.htaccess", "Require all denied\nDeny from all\n");
                if (!file_exists("$d/index.html")) @file_put_contents("$d/index.html", '');
            }
            return $dir = $d;
        }
    }
    out(500, ['error' => 'storage', 'msg' => 'السيرفر ما يسمح نحفظ البيانات، تأكد من صلاحيات المجلد']);
}
function f_data(): string { return data_dir() . '/data.json'; }

            'subtitle' => 'مستشفى السيدة خديجة الكبرى (ع) التخصصي للمرأة', 'printMode' => 'auto', 'printer' => '',
            'gap' => 3, 'density' => 10, 'flip' => false, 'offsetX' => 0]];

function load(): array {
    $b = fresh();
    $raw = @file_get_contents(f_data());
    if ($raw === false || $raw === '') return $b;
    $s = json_decode($raw, true);
    if (!is_array($s)) { @copy(f_data(), f_data() . '.broken-' . time()); return $b; }
    $set = array_merge($b['settings'], (array)($s['settings'] ?? []));
    $s = array_merge($b, $s);
    $s['settings'] = $set;
    foreach (['seats', 'guests', 'users', 'sessions', 'fails', 'orgs', 'ofails', 'oform', 'votes', 'teams', 'wa'] as $k) if (!is_array($s[$k] ?? null)) $s[$k] = [];
    if (!is_array($s['settings']['zone'] ?? null)) $s['settings']['zone'] = $b['settings']['zone'];
    // قياس الستكر صار 30×25 ملم (مرة وحدة، بعدها يكدر الأدمن يغيره من الإعدادات)
    if ((int)($s['settings']['szv'] ?? 0) < 2) { $s['settings']['stickerW'] = 30; $s['settings']['stickerH'] = 25; $s['settings']['szv'] = 2; }
    // كراسي ما موجودة بترتيب القاعة الحالي تنشال، وضيوفها يرجعون للقائمة
    foreach (array_keys($s['seats']) as $sid) if (!seat_rc((string)$sid)) {
        $g = $s['seats'][$sid]['g'] ?? '';
        unset($s['seats'][$sid]);
        if ($g !== '' && isset($s['guests'][$g])) $s['guests'][$g]['s'] = null;
    }
    return $s;
}
function save(array &$S, bool $bump = true): void {
    if ($bump) $S['version']++;
    $tmp = f_data() . '.tmp' . getmypid();
    file_put_contents($tmp, json_encode($S, JSON_UNESCAPED_UNICODE));
    rename($tmp, f_data());
    // نسخة احتياطية كل ساعة (آخر 96 نسخة) حتى البيانات ما تضيع
    $bk = data_dir() . '/backups';
    if (is_dir($bk) || @mkdir($bk, 0700, true)) {
        $f = $bk . '/data-' . date('Ymd-H') . '.json';
        if (!file_exists($f)) {
            @copy(f_data(), $f);
            $all = glob($bk . '/data-*.json') ?: [];
            if (count($all) > 96) { sort($all); foreach (array_slice($all, 0, count($all) - 96) as $x) @unlink($x); }
        }
    }
    file_put_contents(f_ver() . '.tmp', (string)$S['version']);
    rename(f_ver() . '.tmp', f_ver());
}

function seat_rc(string $sid): ?array {
    if (!preg_match('/^R(\d{1,2})-(\d{1,2})$/', $sid, $m)) return null;
    $r = (int)$m[1]; $c = (int)$m[2];
    return ($r >= 1 && $r <= ROWS && $c >= 1 && $c <= COLS && seat_ok($r, $c)) ? [$r, $c] : null;
}
/* تقسيم حسب الجنس: القسم B (كراسي 11-20) كله رجال، والصف الأول من القسم A رجال، وباقي A والقسم C كله نساء */
/* كراسي غير موجودة: A10 بالصف 11، و B10 بالصفوف الزوجية (القسم B يتناوب 10 و 9) */
function seat_ok(int $r, int $c): bool { return !(($r === 11 && ($c === 10 || $c === 21)) || ($c === 20 && $r % 2 === 0)); }
function zone_g(int $r, int $c): string { return (($c >= 11 && $c <= 20) || ($c <= 10 && $r === 1)) ? 'm' : 'f'; }

function user_by_pin(array $S, string $pin): ?string {
    foreach ($S['users'] as $uid => $u) if (hash_equals($u['hash'], hash_pin($pin, $u['salt']))) return $uid;

function auth(array $S, string $tok): ?array {
    if ($tok === '' || !isset($S['sessions'][$tok])) return null;
    $uid = $S['sessions'][$tok]['uid'];
    if (!isset($S['users'][$uid])) return null;
    $u = $S['users'][$uid];
    return ['uid' => $uid, 'name' => $u['name'], 'role' => $u['role'], 'token' => $tok,
        'perms' => $u['role'] === 'admin' ? PERMS : clean_perms($u['perms'] ?? DEFAULT_PERMS),
        'task' => $u['task'] ?? '', 'sup' => $u['sup'] ?? null];
}

function target_admin(array $S, array $me): ?string {
    if ($me['role'] === 'admin') return $me['uid'];
    $sup = $me['sup'];
    if ($sup && isset($S['users'][$sup]) && $S['users'][$sup]['role'] === 'admin') return $sup;
    return admins($S)[0] ?? null;
}
function add_job(array &$S, array $me, string $sid, string $name, string $kind = 'guest'): array {
    [$r, $c] = seat_rc($sid);
    $to = target_admin($S, $me);
    $S['jobSeq']++;
    $j = ['id' => $S['jobSeq'], 'n' => $name, 'sec' => sec_of($r), 'lbl' => lbl_of($r, $c), 'r' => $r, 'seat' => $sid,
        't' => now_ms(), 'by' => $me['uid'], 'byName' => $me['name'], 'to' => $to,
        'toName' => $to ? $S['users'][$to]['name'] : '', 'st' => 'new', 'kind' => $kind, 'claimed' => 'queue'];
    $S['jobs'][] = $j;
    if (count($S['jobs']) > 600) $S['jobs'] = array_slice($S['jobs'], -600);
    return $j;
}

    $out = [];

            'seen' => $u['seen'] ?? 0, 'n' => $cnt[$uid] ?? 0];

}

    }

}

const AR_DIGITS = ['٠' => '0', '١' => '1', '٢' => '2', '٣' => '3', '٤' => '4', '٥' => '5', '٦' => '6', '٧' => '7', '٨' => '8', '٩' => '9',
    '۰' => '0', '۱' => '1', '۲' => '2', '۳' => '3', '۴' => '4', '۵' => '5', '۶' => '6', '۷' => '7', '۸' => '8', '۹' => '9'];
function norm_ph(string $v): string {
    $d = preg_replace('/\D/', '', strtr($v, AR_DIGITS));
    if (strpos($d, '00964') === 0) $d = substr($d, 5); elseif (strpos($d, '964') === 0) $d = substr($d, 3);
    if (strlen($d) === 10 && $d[0] === '7') $d = '0' . $d;
    return substr($d, 0, 15);
}
function dist_m(float $a, float $b, float $c, float $d): float {
    $h = sin(deg2rad($c - $a) / 2) ** 2 + cos(deg2rad($a)) * cos(deg2rad($c)) * sin(deg2rad($d - $b) / 2) ** 2;
    return 2 * 6371000 * asin(min(1.0, sqrt($h)));
}
function in_poly(array $P, float $la, float $lo): bool {
    $in = false; $n = count($P);
    for ($i = 0, $j = $n - 1; $i < $n; $j = $i++) {
        $ya = (float)$P[$i][0]; $xa = (float)$P[$i][1]; $yb = (float)$P[$j][0]; $xb = (float)$P[$j][1];
        if ((($ya > $la) !== ($yb > $la)) && ($lo < ($xb - $xa) * ($la - $ya) / ($yb - $ya) + $xa)) $in = !$in;
    }
    return $in;
}
function poly_edge_m(array $P, float $la, float $lo): float {
    $kx = 111320 * cos(deg2rad($la)); $ky = 110574; $best = INF; $n = count($P);
    for ($i = 0, $j = $n - 1; $i < $n; $j = $i++) {
        $ax = ((float)$P[$j][1] - $lo) * $kx; $ay = ((float)$P[$j][0] - $la) * $ky;
        $bx = ((float)$P[$i][1] - $lo) * $kx; $by = ((float)$P[$i][0] - $la) * $ky;
        $dx = $bx - $ax; $dy = $by - $ay; $l = $dx * $dx + $dy * $dy;
        $t = $l > 0 ? max(0.0, min(1.0, -($ax * $dx + $ay * $dy) / $l)) : 0.0;
        $best = min($best, hypot($ax + $t * $dx, $ay + $t * $dy));
    }
    return $best;
}

function org_clean(array $b, array $o): array {
    foreach ([['n', 80], ['area', 80], ['uni', 120], ['spec', 120], ['field', 220], ['exp', 400], ['avail', 60], ['avt', 80], ['soc', 160], ['note', 220]] as [$k, $n])
        if (!isset($o[$k]) || array_key_exists($k, $b)) $o[$k] = mb_substr(trim((string)preg_replace('/\s+/u', ' ', (string)($b[$k] ?? ''))), 0, $n);
    if (!isset($o['g']) || array_key_exists('g', $b)) $o['g'] = ($b['g'] ?? '') === 'm' ? 'm' : 'f';
    foreach (['avf', 'avto'] as $k) if (!array_key_exists($k, $o) || array_key_exists($k, $b)) { $h = $b[$k] ?? null; $o[$k] = (is_numeric($h) && (int)$h >= 0 && (int)$h <= 24) ? (int)$h : null; }
    if (!isset($o['shift']) || array_key_exists('shift', $b)) $o['shift'] = in_array($b['shift'] ?? '', ['a', 'b'], true) ? $b['shift'] : '';
    if (!isset($o['ph']) || array_key_exists('ph', $b)) $o['ph'] = norm_ph((string)($b['ph'] ?? ''));
    if (!array_key_exists('age', $o) || array_key_exists('age', $b)) {
        $a = (int)preg_replace('/\D/', '', strtr((string)($b['age'] ?? ''), AR_DIGITS));
        $o['age'] = ($a >= 10 && $a <= 90) ? $a : null;
    }
    return $o;
}
// مقارنة التصويت مع بيانات المنظم (تنحسب كل مرة تنفتح القائمة)
function vote_cmp(array $S, array $x): array {
    $v = null; $vp = '';
    foreach ([$x['ph'] ?? '', $x['vph'] ?? ''] as $p) if ($p !== '' && isset($S['votes'][$p])) { if (!$v || $S['votes'][$p]['t'] > $v['t']) { $v = $S['votes'][$p]; $vp = $p; } }
    if (!$v) return ['vote' => null];
    return ['vote' => ['n' => $v['n'], 'shift' => $v['shift'], 't' => $v['t'], 'c' => $v['c'] ?? 1, 'ph' => $vp,
        'how' => $vp === ($x['ph'] ?? '') ? 'phone' : 'name', 'nameOk' => same_name($v['n'], $x['n'] ?? '')]];
}
function votes_loose(array $S): array {
    $own = [];
    foreach ($S['orgs'] as $o) { if (($o['ph'] ?? '') !== '') $own[$o['ph']] = 1; if (($o['vph'] ?? '') !== '') $own[$o['vph']] = 1; }
    $out = [];
    foreach ($S['votes'] as $p => $v) if (!isset($own[$p])) $out[] = ['ph' => (string)$p] + $v;
    return $out;
}
function orgs_out(array $S): array {
    $out = [];
    foreach ($S['orgs'] as $id => $x) {
        $uid = $x['uid'] ?? null;
        $u = ($uid && isset($S['users'][$uid])) ? $S['users'][$uid] : null;
        $out[] = array_merge(['id' => $id], $x, ['uid' => $u ? $uid : null, 'pin' => $u ? ($x['pin'] ?? null) : null,
            'role' => $u['role'] ?? null, 'perms' => $u ? ($u['role'] === 'admin' ? PERMS : clean_perms($u['perms'] ?? DEFAULT_PERMS)) : [],
            'task' => $u['task'] ?? '', 'sup' => $u['sup'] ?? null, 'seen' => $u['seen'] ?? 0, 'att' => obj((array)($x['att'] ?? []))], vote_cmp($S, $x));
    }
    return $out;
}
/* الفرق: كل فريق بيه أعضاء من المنظمين، لون، ومكانه على مخطط القاعة (دائرة) */
function teams_out(array $S): array {
    $out = [];
    foreach ($S['teams'] as $id => $t) {
        $m = array_values(array_filter((array)($t['m'] ?? []), fn($x) => isset($S['orgs'][$x])));
        $out[] = ['id' => (string)$id] + $t + ['m' => $m];
    }
    usort($out, fn($a, $b) => ($a['ct'] ?? 0) <=> ($b['ct'] ?? 0));
    return $out;
}

    return $o;
}

    foreach ($S['orgs'] as &$o) if (($o['uid'] ?? null) === $uid) { $o['uid'] = null; $o['pin'] = null; }
    unset($o);
}

/* ----- روابط المنظمين العامة: تحديث المعلومات والتصويت على وقت الحضور (بدون دخول) ----- */
function org_rate(array &$S): void {
    $ip = client_ip(); $now = time();
    $f = $S['oform'][$ip] ?? [0, $now];
    if ($now - $f[1] > 3600) $f = [0, $now];
    if ($f[0] >= 60) out(429, ['error' => 'wait', 'msg' => 'محاولات كثيرة من هذا الجهاز، حاول بعد قليل']);
    $f[0]++; $S['oform'][$ip] = $f;
    foreach ($S['oform'] as $k => $x) if ($now - ($x[1] ?? 0) > 7200) unset($S['oform'][$k]);
}
function nname(string $x): string {
    $x = preg_replace(['/[إأآ]/u', '/ة/u', '/ى/u', '/ؤ/u', '/ئ/u', '/[\x{064B}-\x{0652}\x{0640}]/u', '/[^\p{Arabic}\p{L}\s]/u', '/\s+/u'], ['ا', 'ه', 'ي', 'و', 'ي', '', ' ', ' '], trim($x));
    $x = preg_replace('/(^|\s)عبد\s+/u', '$1عبد', (string)$x);
    return trim((string)$x);
}
// نفس الشخص إذا أول اسمين متطابقين والاسم الأقصر جزء من الأطول
function same_name(string $a, string $b): bool {
    $x = explode(' ', nname($a)); $y = explode(' ', nname($b));
    if (count($x) < 2 || count($y) < 2) return false;
    $n = min(count($x), count($y));
    if ($n < 3 && count($x) !== count($y)) return false; // الاسم الثنائي لازم يطابق بالكامل
    return array_slice($x, 0, $n) === array_slice($y, 0, $n);
}
function org_by_name(array $S, string $name, string $skipPh = ''): ?string {
    $hit = [];
    foreach ($S['orgs'] as $k => $o) if (same_name($o['n'] ?? '', $name)) $hit[] = (string)$k;
    return count($hit) === 1 ? $hit[0] : null;
}
